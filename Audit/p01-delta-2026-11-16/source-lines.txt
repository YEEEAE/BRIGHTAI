### src/pages/app/api/secure-chat/messages.ts
1: import type { APIRoute } from 'astro';
2: import { errorResponse, privateHeaders, sendMessage, cancelMessage } from '../../../../server/secure-chat/service.mjs';
3: import { KernelError } from '../../../../server/kernel/core.mjs';
4: export const prerender = false;
5: 
6: async function bodyFor(request: Request, keys: string[]) {
7:   if (request.headers.get('origin') !== new URL(request.url).origin) throw new KernelError(403,'ORIGIN_REJECTED','مصدر الطلب غير مسموح.');
8:   if (request.headers.get('content-type')?.split(';')[0].trim() !== 'application/json') throw new KernelError(415,'INVALID_BODY','صيغة الطلب غير مدعومة.');
9:   const reader=request.body?.getReader();
10:   if (!reader) throw new KernelError(400,'INVALID_BODY','الطلب فارغ.');
11:   const decoder=new TextDecoder();
12:   let raw='',bytes=0;
13:   try {
14:     while (true) {
15:       const {done,value}=await reader.read();
16:       if (done) break;
17:       bytes+=value.byteLength;
18:       if (bytes>131072) { await reader.cancel(); throw new KernelError(413,'BODY_TOO_LARGE','الطلب أكبر من الحد التقني المسموح.'); }
19:       raw+=decoder.decode(value,{stream:true});
20:     }
21:     raw+=decoder.decode();
22:   } finally { reader.releaseLock(); }
23:   let body;
24:   try { body=JSON.parse(raw); } catch { throw new KernelError(400,'INVALID_BODY','تعذر قراءة الطلب.'); }
25:   if (!body || typeof body!=='object' || Array.isArray(body) || Object.keys(body).some(key=>!keys.includes(key))) throw new KernelError(400,'INVALID_BODY','حقول الطلب غير مدعومة.');
26:   return body;
27: }
28: export const POST: APIRoute = async ({request}) => {
29:   try {
30:     const body=await bodyFor(request,['requestId','modelId','message','history','syntheticAcknowledged']);
31:     return Response.json(await sendMessage(request,body),{headers:privateHeaders});
32:   } catch(error) { return errorResponse(error); }
33: };
34: export const DELETE: APIRoute = async ({request}) => {
35:   try {
36:     const body=await bodyFor(request,['requestId']);
37:     return Response.json(await cancelMessage(request,body.requestId),{headers:privateHeaders});
38:   } catch(error) { return errorResponse(error); }
39: };
